// Fix orphan issues (projectId null or identifier null) in v2 company.

import path from "node:path";
import { fileURLToPath } from "node:url";
import { createRequire } from "node:module";

const thisDir = path.dirname(fileURLToPath(import.meta.url));
const dbRequire = createRequire(path.resolve(thisDir, "..", "packages", "db", "package.json"));
// eslint-disable-next-line @typescript-eslint/no-explicit-any
const postgres = dbRequire("postgres") as (url: string, opts?: any) => any;

const DB_URL = process.env.PAPERCLIP_DB_URL ?? "postgres://paperclip:paperclip@127.0.0.1:54329/paperclip";
const COMPANY_ID = "68892580-9963-4506-a987-90be8d8ddb26";
const DEFAULT_PROJECT = "39f55e51-9245-4607-b4b8-0ec5d8eca96f";

async function main(): Promise<void> {
  const sql = postgres(DB_URL, { max: 1 });
  try {
    const sample = await sql`
      SELECT id, title, status, lifecycle_state, project_id, identifier, issue_number, origin_kind
      FROM issues
      WHERE company_id = ${COMPANY_ID}
        AND (project_id IS NULL OR identifier IS NULL)
      LIMIT 5
    `;
    console.log("Sample orphans:", JSON.stringify(sample, null, 2));

    await sql`
      UPDATE issues
      SET project_id = ${DEFAULT_PROJECT}, updated_at = NOW()
      WHERE company_id = ${COMPANY_ID} AND project_id IS NULL
    `;

    const noIdent = await sql`
      SELECT id FROM issues
      WHERE company_id = ${COMPANY_ID} AND identifier IS NULL
      ORDER BY created_at
    `;
    console.log(`Issues missing identifier: ${noIdent.length}`);

    for (const row of noIdent) {
      const counterRow = await sql`
        UPDATE companies SET issue_counter = issue_counter + 1
        WHERE id = ${COMPANY_ID}
        RETURNING issue_counter, issue_prefix
      `;
      const { issue_counter: n, issue_prefix: prefix } = counterRow[0];
      await sql`
        UPDATE issues
        SET identifier = ${`${prefix}-${n}`},
            issue_number = ${n},
            updated_at = NOW()
        WHERE id = ${row.id}
      `;
    }

    const after = await sql`
      SELECT count(*)::int AS n FROM issues
      WHERE company_id = ${COMPANY_ID}
        AND (project_id IS NULL OR identifier IS NULL)
    `;
    console.log("Remaining broken:", JSON.stringify(after, null, 2));
  } finally {
    await sql.end({ timeout: 5 });
  }
}

main().catch((err: unknown) => { console.error(err); process.exit(1); });
```

Idempotency notes:
- The project_id UPDATE is already idempotent (the `WHERE project_id IS NULL` guard means the 2nd run matches 0 rows).
- The identifier UPDATE loop is idempotent BY ACCIDENT: because the outer SELECT is guarded by `identifier IS NULL`, a 2nd run finds 0 rows and doesn't increment the counter. This is the "guard that only targets rows where identifier IS NULL" mentioned in CONTEXT.md specifics (line 141).
- BUT: if the loop is INTERRUPTED mid-way (e.g. process killed after bumping counter but before UPDATE of the issue row), you'd have a bumped counter with no row assigned. The counter is monotonic (UPDATE ... SET counter = counter + 1) so this just produces a gap in identifiers — not a correctness bug. Document this.

       async function main(): Promise<void> {
         const sql = postgres(DB_URL, { max: 1 });
         const report: Record<string, unknown> = { dryRun: DRY_RUN, companyId: COMPANY_ID };
         try {
           // Phase A: sample orphans (diagnostic, always runs)
           const sample = await sql`
             SELECT id, title, status, lifecycle_state, project_id, identifier, issue_number, origin_kind, created_at
             FROM issues
             WHERE company_id = ${COMPANY_ID}
               AND (project_id IS NULL OR identifier IS NULL)
             ORDER BY created_at ASC
             LIMIT 10
           `;
           report.sampleOrphans = sample;
           console.log(`[backfill] sample orphans (up to 10):`, JSON.stringify(sample, null, 2));

           // Phase B: resolve default project UUID by name (not by baked-in UUID)
           const projRows = await sql`
             SELECT id FROM projects
             WHERE company_id = ${COMPANY_ID} AND name = ${DEFAULT_PROJECT_NAME}
             LIMIT 1
           `;
           if (projRows.length === 0) {
             throw new Error(`Default project "${DEFAULT_PROJECT_NAME}" not found for company ${COMPANY_ID}. Aborting.`);
           }
           const defaultProjectId = projRows[0].id;
           report.defaultProjectId = defaultProjectId;

           // Phase C: fill null project_id (idempotent via WHERE guard)
           const nullProjBefore = await sql`
             SELECT count(*)::int AS n FROM issues
             WHERE company_id = ${COMPANY_ID} AND project_id IS NULL
           `;
           report.nullProjectIdBefore = nullProjBefore[0].n;
           if (!DRY_RUN && nullProjBefore[0].n > 0) {
             await sql`
               UPDATE issues
               SET project_id = ${defaultProjectId}, updated_at = NOW()
               WHERE company_id = ${COMPANY_ID} AND project_id IS NULL
             `;
           }
           const nullProjAfter = await sql`
             SELECT count(*)::int AS n FROM issues
             WHERE company_id = ${COMPANY_ID} AND project_id IS NULL
           `;
           report.nullProjectIdAfter = nullProjAfter[0].n;
           console.log(`[backfill] project_id: before=${nullProjBefore[0].n} after=${nullProjAfter[0].n} (dry-run=${DRY_RUN})`);

           // Phase D: assign identifier for rows where identifier IS NULL
           const noIdent = await sql`
             SELECT id FROM issues
             WHERE company_id = ${COMPANY_ID} AND identifier IS NULL
             ORDER BY created_at ASC
           `;
           report.nullIdentifierBefore = noIdent.length;
           console.log(`[backfill] identifier: ${noIdent.length} row(s) to assign (dry-run=${DRY_RUN})`);

           const assigned: Array<{ id: string; identifier: string; issueNumber: number }> = [];
           if (!DRY_RUN) {
             for (const row of noIdent) {
               // Bump counter + return prefix in a single UPDATE ... RETURNING (monotonic).
               const counterRow = await sql`
                 UPDATE companies SET issue_counter = issue_counter + 1
                 WHERE id = ${COMPANY_ID}
                 RETURNING issue_counter, issue_prefix
               `;
               const { issue_counter: n, issue_prefix: prefix } = counterRow[0];
               const identifier = `${prefix}-${n}`;
               await sql`
                 UPDATE issues
                 SET identifier = ${identifier},
                     issue_number = ${n},
                     updated_at = NOW()
                 WHERE id = ${row.id} AND identifier IS NULL
               `;
               // The extra `AND identifier IS NULL` on the UPDATE is belt-and-braces —
               // protects against a concurrent writer between SELECT and UPDATE.
               assigned.push({ id: row.id, identifier, issueNumber: n });
             }
           }
           report.assigned = assigned;

           // Phase E: final state
           const after = await sql`
             SELECT count(*)::int AS n FROM issues
             WHERE company_id = ${COMPANY_ID}
               AND (project_id IS NULL OR identifier IS NULL)
           `;
           report.remainingBroken = after[0].n;
           console.log(`[backfill] remaining broken rows: ${after[0].n}`);
           console.log(`[backfill] report:`, JSON.stringify(report, null, 2));
           if (DRY_RUN) {
             console.log(`[backfill] DRY RUN — no writes performed. Re-run without --dry-run to apply.`);
           }
         } finally {
           await sql.end({ timeout: 5 });
         }
       }

       main().catch((err: unknown) => { console.error(err); process.exit(1); });
       ```
